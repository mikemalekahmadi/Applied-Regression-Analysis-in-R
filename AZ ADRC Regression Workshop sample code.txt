library(ggplot2)

## Importing data from a Microsoft Excel spreadsheet
library(readxl)
workshop_dataset<-read_excel("workshop_dataset.xlsx")
View(npi_dataset)
summary(workshop_dataset)

# Histogram for inspection of the dependent variable
hist(workshop_dataset$animal_fluency)
hist(workshop_dataset$geri_dep_scale)

# QQ Plot for inspection of the dependent variable
qqnorm(workshop_dataset$animal_fluency)
qqnorm(workshop_dataset$geri_dep_scale)

# Shapiro-Wilk test for the normality assumption of the dependent variable
shapiro.test(workshop_dataset$animal_fluency)
shapiro.test(workshop_dataset$geri_dep_scale)

# Simple linear regression model for CDR-SOB as a predictor of Animal Fluency performance
cdr_sob_animal<-lm(animal_fluency~cdr_sob, data=workshop_dataset)
summary(cdr_sob_animal)

# Multivariable linear regression model for CDR-SOB as a predictor of Animal Fluency performance
#after adjusting for Geriatric Depression Scale and MMSE scores. 
cdr_sob_animal_adjusted<-lm(animal_fluency~geri_dep_scale+mmse+cdr_sob, data=workshop_dataset)
summary(cdr_sob_animal_adjusted)

#Spearman correlation for CDR-SOB and MMSE
cor(workshop_dataset$cdr_sob, workshop_dataset$mmse, method=“spearman”, use=“complete.obs”)

# Comparison of Akaike's Information Criteria (AIC) values between simple and multivariable
# linear regression models
AIC(cdr_sob_animal)
AIC(cdr_sob_animal_adjusted)

#Linear and robust regression models using the Judgment of Line Orientation test as a predictor of the Alzheimer's Questionnaire
library(robust)
library(robustbase)
hist(workshop_dataset$faq)
qqnorm(workshop_dataset$faq)
linear_animal_fluency_faq<-lm(faq~animal_fluency, data = workshop_dataset)
robust_animal_fluency_faq<-lmRob(faq~animal_fluency, data = workshop_dataset)

#View results of models
summary(linear_animal_fluency_faq)
summary(robust_animal_fluency_faq)
